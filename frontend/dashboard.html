        Dim conn As New SqlConnection("Server=yourserver;Database=cars;User Id=...;Password=...;")
        Dim cmd As New SqlCommand("SELECT TOP 100 id, plate, owner, noted FROM scans ORDER BY created_at DESC", conn)
        conn.Open()
        Dim rdr As SqlDataReader = cmd.ExecuteReader()
        While rdr.Read()

        End While
        conn.Close()
